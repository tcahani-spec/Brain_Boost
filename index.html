async function fetchFreshQuestionsFromGemini(profile) { const apiKey = localStorage.getItem('gemini_api_key'); if (!apiKey) { updateStatusBadge("Local Bank (No Key)"); return; } // Cancel any prior in-flight fetch if (abortController) abortController.abort(); abortController = new AbortController(); updateStatusBadge("Syncing AI..."); const grade = profile === 'william' ? "2nd Grade" : "4th Grade"; const prompt = `Generate 5 multiple-choice trivia questions for a ${grade} student. Subjects: Math, Science, Spelling, Engineering, and AI. Provide output ONLY as a JSON array of objects with keys: "q" (string), "options" (array of 4 strings), and "answer" (index 0-3).`; try { // 1. Updated endpoint to standard gemini-2.0-flash const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${apiKey}`, { method: 'POST', signal: abortController.signal, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }], generationConfig: { responseMimeType: "application/json" } }) }); if (!response.ok) { const errDetails = await response.json().catch(() => ({})); throw new Error(`HTTP ${response.status}: ${errDetails.error?.message || response.statusText}`); } const data = await response.json(); let rawText = data.candidates?.[0]?.content?.parts?.[0]?.text || ""; // 2. Clean markdown fences (```json ... ```) before parsing rawText = rawText.replace(/```json/gi, '').replace(/```/g, '').trim(); const parsedQuestions = JSON.parse(rawText); if (Array.isArray(parsedQuestions) && parsedQuestions.length >= 5) { STATIC_BANKS[profile] = parsedQuestions.slice(0, 5); updateStatusBadge("AI Questions Ready!"); } else { throw new Error("Invalid JSON structure received"); } } catch (err) { if (err.name === 'AbortError') { console.log("Fetch aborted due to screen change."); } else { console.error("Gemini API Error Detail:", err); // 3. Display the real error on the status badge so you can diagnose instantly const displayErr = err.message.length > 22 ? err.message.slice(0, 22) + "..." : err.message; updateStatusBadge(`Err: ${displayErr}`); } } }
